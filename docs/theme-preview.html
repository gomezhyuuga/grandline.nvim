import { useState, useCallback } from 'react';
import type { CrewMember, Bounty } from './types';
interface CrewCardProps {
member: CrewMember;
onBountyUpdate?: (bounty: Bounty) => void;
}
export const CrewCard: React.FC<CrewCardProps> = ({ member, onBountyUpdate }) => {
const [isAwakened, setIsAwakened] = useState<boolean>(false);
const handleGearFifth = useCallback(() => {
setIsAwakened(true);
onBountyUpdate?.({ amount: 3_000_000_000, currency: 'beli' });
}, [onBountyUpdate]);
return (
<div className={`crew-card ${isAwakened ? 'awakened' : ''}`}>
<h2>{member.name}</h2>
<span className="epithet">{member.epithet}</span>
<p>Bounty: {member.bounty.toLocaleString()} Beli</p>
<button onClick={handleGearFifth} disabled={isAwakened}>
{isAwakened ? 'Nika Awakened!' : 'Activate Gear Fifth'}
</button>
</div>
);
};
export default CrewCard;
